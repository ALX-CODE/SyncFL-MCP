cmake_minimum_required(VERSION 3.24)
project(ALXFLHostBridge VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This research prototype requires Windows x64 and MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(FL_SDK_ROOT "" CACHE PATH "Locally extracted official fpsdk_20210311b.zip root")
set(sdk_headers "${FL_SDK_ROOT}/source/c")
foreach(pair IN ITEMS
    "fp_plugclass.h|d04e33ac267801da0d5e8492271f8fc11ce4182dca7b8cfa1e6ec49226275444"
    "fp_def.h|b291de22a41940b6d89fab785a4d7e26bb429e96f63e3ed358e901add4c0bda8")
  string(REPLACE "|" ";" entry "${pair}")
  list(GET entry 0 header)
  list(GET entry 1 expected)
  if(NOT EXISTS "${sdk_headers}/${header}")
    message(FATAL_ERROR "Set FL_SDK_ROOT to the extracted official SDK root")
  endif()
  file(SHA256 "${sdk_headers}/${header}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Unreviewed SDK header: ${header}. Requalify its ABI before building.")
  endif()
endforeach()
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options(/W4 /WX /permissive- /EHsc /sdl /guard:cf)
add_link_options(/DYNAMICBASE /NXCOMPAT /guard:cf)
add_library(host-ipc STATIC src/ipc.cpp)
target_include_directories(host-ipc PUBLIC include)
target_link_libraries(host-ipc PUBLIC advapi32 bcrypt)
add_library(alx-fl-host-bridge SHARED src/plugin.cpp)
target_include_directories(alx-fl-host-bridge SYSTEM PRIVATE "${sdk_headers}")
target_link_libraries(alx-fl-host-bridge PRIVATE host-ipc)
set_target_properties(alx-fl-host-bridge PROPERTIES OUTPUT_NAME "ALX FL Host Bridge")
add_executable(alx-fl-host src/cli.cpp)
target_link_libraries(alx-fl-host PRIVATE host-ipc)
add_executable(host-bridge-tests tests/fake_host.cpp)
target_include_directories(host-bridge-tests SYSTEM PRIVATE "${sdk_headers}")
target_link_libraries(host-bridge-tests PRIVATE host-ipc)
enable_testing()
add_test(NAME fake_host_roundtrip COMMAND host-bridge-tests
  "$<TARGET_FILE:alx-fl-host-bridge>" "$<TARGET_FILE:alx-fl-host>")
set_tests_properties(fake_host_roundtrip PROPERTIES TIMEOUT 25)
add_test(NAME cli_rejects_bad_args COMMAND alx-fl-host --pid not-a-number ping)
set_tests_properties(cli_rejects_bad_args PROPERTIES WILL_FAIL TRUE)
get_filename_component(msvc_tools "${CMAKE_LINKER}" DIRECTORY)
add_test(NAME no_desktop_input_imports COMMAND "${CMAKE_COMMAND}"
  "-DDUMPBIN=${msvc_tools}/dumpbin.exe"
  "-DBINARY=$<TARGET_FILE:alx-fl-host-bridge>"
  "-DSOURCE_ROOT=${CMAKE_CURRENT_SOURCE_DIR}"
  -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_boundary.cmake")
