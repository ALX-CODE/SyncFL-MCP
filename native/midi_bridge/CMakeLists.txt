cmake_minimum_required(VERSION 3.24)
project(ALXFLBridge VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "ALX FL Bridge requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
# C++/WinRT otherwise embeds the builder's absolute source paths in exceptions.
# Keep HRESULT diagnostics without publishing a developer's home directory.
add_compile_definitions(WINRT_NO_SOURCE_LOCATION)
enable_testing()

set(MIDI_SDK_ROOT "${CMAKE_BINARY_DIR}/deps/midi-sdk" CACHE PATH "Extracted official Windows.Devices.Midi2 package")
set(MIDI_PROJECTION_ROOT "${CMAKE_BINARY_DIR}/deps/generated" CACHE PATH "Generated C++/WinRT projection")
if(NOT EXISTS "${MIDI_PROJECTION_ROOT}/winrt/Windows.Devices.Midi2.h")
  message(FATAL_ERROR "Missing SDK projections. Run build.ps1 -FetchDependencies first.")
endif()
  add_executable(alx-midi-service src/service_bridge.cpp)
  target_compile_definitions(alx-midi-service PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_include_directories(alx-midi-service SYSTEM PRIVATE "${MIDI_PROJECTION_ROOT}" "${CMAKE_BINARY_DIR}/deps/cppwinrt/include")
  target_link_libraries(alx-midi-service PRIVATE windowsapp winmm)
  target_compile_options(alx-midi-service PRIVATE /W4 /WX /permissive- /bigobj /sdl /guard:cf)
  target_link_options(alx-midi-service PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
  add_executable(service_roundtrip tests/service_roundtrip.cpp)
  target_compile_definitions(service_roundtrip PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(service_roundtrip PRIVATE /W4 /WX /permissive- /bigobj /sdl /guard:cf)
  target_include_directories(service_roundtrip SYSTEM PRIVATE "${MIDI_PROJECTION_ROOT}")
  target_link_libraries(service_roundtrip PRIVATE winmm windowsapp)
  target_link_options(service_roundtrip PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
  add_custom_command(TARGET alx-midi-service POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${MIDI_SDK_ROOT}/runtimes/win-x64/native/Windows.Devices.Midi2.dll" "$<TARGET_FILE_DIR:alx-midi-service>"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${MIDI_SDK_ROOT}/runtimes/win-x64/native/Windows.Devices.Midi2.pri" "$<TARGET_FILE_DIR:alx-midi-service>")
add_test(NAME service_status COMMAND alx-midi-service status --json)
set_tests_properties(service_status PROPERTIES SKIP_RETURN_CODE 3)
add_test(NAME service_roundtrip COMMAND service_roundtrip)
set_tests_properties(service_roundtrip PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 25)
add_test(NAME rejects_invalid_command COMMAND alx-midi-service invalid)
set_tests_properties(rejects_invalid_command PROPERTIES WILL_FAIL TRUE)
